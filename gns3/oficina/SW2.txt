! ================================================================
! SW2 — Switch Distribución L3 Oficina (Cisco vios_l2)
! HSRP Standby (priority 100) — Spanning-Tree Root Secondary
! ================================================================
configure terminal

! BLOQUE 1: BÁSICA
hostname SW2
ip domain-name surnet.local
ip routing
service password-encryption
enable secret SW2
username SW2 privilege 15 secret SW2
no ip domain-lookup

banner motd ^
ACCESO RESTRINGIDO - Solo personal autorizado
^

crypto key generate rsa modulus 2048
ip ssh version 2

line console 0
 login local
 logging synchronous
 exec-timeout 5 0
 exit

line vty 0 4
 login local
 transport input ssh
 logging synchronous
 exec-timeout 5 0
 exit

! BLOQUE 2: VLANs
vlan 10
 name RRHH
vlan 20
 name Ventas
vlan 30
 name Administracion
vlan 40
 name Gerencia
vlan 50
 name Visitas
vlan 99
 name Servicios
 exit

! BLOQUE 3: INTERFACES L3
interface Loopback0
 description Loopback - Router-ID OSPF
 ip address 10.0.1.22 255.255.255.255
 no shutdown
 exit

interface GigabitEthernet0/0
 description P2P hacia R1-G0/0
 no switchport
 ip address 192.168.10.22 255.255.255.252
 no shutdown
 exit

 exit

interface GigabitEthernet2/1
 no switchport

 no shutdown
 exit

interface GigabitEthernet2/2
 no switchport

 no shutdown
 exit

interface Port-channel1
 description Inter-distribution SW1-SW2 (LACP)
 no switchport
 ip address 192.168.10.34 255.255.255.252
 no shutdown
 exit

! BLOQUE 4: TRUNKS Y ACCESO
interface GigabitEthernet0/2
 description Trunk hacia SW4
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40,50,99
 no shutdown
 exit

interface GigabitEthernet2/0
 description Trunk hacia SW3
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40,50,99
 no shutdown
 exit

interface GigabitEthernet1/0
 description Puerto Server-Oficina (NIC2)
 switchport mode access
 switchport access vlan 99

 spanning-tree bpduguard enable
 no shutdown
 exit

 exit

 exit

! BLOQUE 7: SPANNING-TREE (SW2 = root secondary todas las VLANs Oficina)

! BLOQUE 8: ACL SEGURIDAD VISITAS

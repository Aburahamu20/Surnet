! ================================================================
! SW6 — Switch Distribución L3 Bodega (Cisco vios_l2)
! HSRP Standby (priority 100) — Spanning-Tree Root Secondary
! ================================================================
configure terminal

! BLOQUE 1: BÁSICA
hostname SW6
ip domain-name surnet.local
ip routing
service password-encryption
enable secret SW6
username SW6 privilege 15 secret SW6
no ip domain-lookup

banner motd ^
ACCESO RESTRINGIDO - Solo personal autorizado
^

crypto key generate rsa modulus 2048
ip ssh version 2

line console 0
 login local
 logging synchronous
 exec-timeout 5 0
 exit

line vty 0 4
 login local
 transport input ssh
 logging synchronous
 exec-timeout 5 0
 exit

! BLOQUE 2: VLANs
vlan 30
 name Administracion
vlan 60
 name Operaciones
vlan 70
 name Bodega
vlan 99
 name Servicios
 exit

! BLOQUE 3: INTERFACES L3
interface Loopback0
 description Loopback - Router-ID OSPF
 ip address 10.0.2.22 255.255.255.255
 no shutdown
 exit

interface GigabitEthernet0/0
 description P2P hacia R3 G0/1
 no switchport
 ip address 192.168.20.14 255.255.255.252
 no shutdown
 exit

interface GigabitEthernet0/1
 description P2P hacia R4 G0/1 (cruzado)
 no switchport
 ip address 192.168.20.22 255.255.255.252
 no shutdown
 exit

interface GigabitEthernet2/1
 no switchport
 channel-group 1 mode active
 no shutdown
 exit

interface GigabitEthernet2/2
 no switchport
 channel-group 1 mode active
 no shutdown
 exit

interface Port-channel1
 description Inter-distribution SW5-SW6 (LACP)
 no switchport
 ip address 192.168.20.26 255.255.255.252
 no shutdown
 exit

! BLOQUE 4: TRUNKS Y ACCESO
interface GigabitEthernet0/2
 description Trunk hacia SW8
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 30,60,70,99
 no shutdown
 exit

interface GigabitEthernet2/0
 description Trunk hacia SW7
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 30,60,70,99
 no shutdown
 exit

interface GigabitEthernet1/0
 description Puerto Server-Bodega (NIC2)

 no shutdown

! ================================================================
! SW8 — Switch Acceso L2 Bodega (Cisco vios_l2)
! ================================================================
configure terminal

hostname SW8
ip domain-name surnet.local
service password-encryption
enable secret SW8
username SW8 privilege 15 secret SW8
no ip domain-lookup

banner motd ^
ACCESO RESTRINGIDO - Solo personal autorizado
^

crypto key generate rsa modulus 2048
ip ssh version 2

line console 0
 login local
 logging synchronous
 exec-timeout 5 0
 exit

line vty 0 4
 login local
 transport input ssh
 logging synchronous
 exec-timeout 5 0
 exit
